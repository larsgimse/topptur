function visProfil(gpxLayer, tittel, gpxFil) { const d = gpxLayer.get_distance(), e = gpxLayer.get_elevation_gain(), h = gpxLayer.get_elevation_max(), minH = gpxLayer.get_elevation_min(), stats = gpxLayer.get_elevation_data(); const dato = parseDatoFraFil(gpxFil); let tidTR = "N/A", tidTilTopp = "N/A"; const startTid = gpxLayer.get_start_time(), sluttTid = gpxLayer.get_end_time(); // Finn koordinatet for topppunktet i GPX-filen const punkter = gpxLayer.getLayers()[0].getLatLngs(); let toppKoordinat = null; let høyesteMålt = -9999; for (let p of punkter) { let gjeldendeHøyde = (p.meta && p.meta.ele) ? p.meta.ele : h; if (gjeldendeHøyde > høyesteMålt) { høyesteMålt = gjeldendeHøyde; toppKoordinat = p; } } if (startTid && sluttTid) { const diffMs = sluttTid - startTid; tidTR = `${Math.floor(diffMs / 3600000)}t ${Math.floor((diffMs % 3600000) / 60000)}min`; let toppTid = null; for (let p of punkter) { if (p.meta && p.meta.ele >= h - 0.5) { toppTid = p.meta.time; break; } } if (toppTid) { const dTopp = toppTid - startTid; tidTilTopp = `${Math.floor(dTopp / 3600000)}t ${Math.floor((dTopp % 3600000) / 60000)}min`; } } // Generer Yr-lenke via søkefunksjonen deres som takler koordinater perfekt let yrLenkeHtml = ''; if (toppKoordinat) { const lat = toppKoordinat.lat; const lon = toppKoordinat.lng; const yrUrl = `https://www.yr.no/nb/søk?q=${lat},${lon}`; yrLenkeHtml = `🌤️ Værvarsel på Yr`; } const popupContent = document.createElement('div'); popupContent.innerHTML = ` ${tittel}
${dato ? `${dato}
` : ''}
📏 ${(d/1000).toFixed(1)} km | 📈 ${e.toFixed(0)}m | 🏔️ ${h.toFixed(0)} moh
⏱️ Til topp: ${tidTilTopp} | ⌛ T/R: ${tidTR}
${yrLenkeHtml}`; gpxLayer.bindPopup(popupContent).openPopup(); setTimeout(() => { const canvas = document.getElementById('elevationChart'); if (!canvas) return; const ctx = canvas.getContext('2d'), w = canvas.width, ht = canvas.height, pad = 5, maxD = stats[stats.length - 1][0], rH = h - minH || 1; ctx.beginPath(); ctx.strokeStyle = 'var(--strava-orange)'; ctx.fillStyle = 'rgba(252, 76, 2, 0.15)'; ctx.moveTo(pad, ht - pad); stats.forEach(([dist, elev]) => ctx.lineTo(pad + (dist / maxD) * (w - pad * 2), (ht - pad) - ((elev - minH) / rH) * (ht - pad * 2))); ctx.lineTo(w - pad, ht - pad); ctx.closePath(); ctx.fill(); ctx.stroke(); }, 100); }
